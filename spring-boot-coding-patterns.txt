@Override
public List<CommunityShortInfo> getCommunityDetailsByWorkspace(Workspace workspace) {

    List<Community> communities =
            communityRepository.findByWorkspaceId(workspace.getId());

    List<CommunityShortInfo> communityList = new ArrayList<>();

    for (Community community : communities) {

        List<Campaign> campaigns =
                campaignRepository.findByCommunityId(community.getId());

        List<CampaignDataInfo> campaignList = new ArrayList<>();

        for (Campaign campaign : campaigns) {

            CampaignDataInfo campaignInfo = new CampaignDataInfo(
                    campaign.getCommunityId(),
                    campaign.getCampaignGroupId(),
                    campaign.isDisabled(),
                    campaign.getName()
            );

            campaignList.add(campaignInfo);
        }

        CommunityShortInfo communityInfo =
                new CommunityShortInfo(community.getTitle(), campaignList);

        communityList.add(communityInfo);
    }

    return communityList;
}

Using builder function and stream api:

@Override
public List<CommunityShortInfo> getCommunityDetailsByWorkspace(Workspace workspace) {

    List<Community> communities = communityRepository.findByWorkspaceId(workspace.getId());

    List<CommunityShortInfo> communityList = new ArrayList<>();

    for (Community community : communities) {

        List<Campaign> campaigns = campaignRepository.findAllByCommunityId(community.getId());

        // Build campaign DTOs
        List<CampaignDataInfo> campaignList = campaigns.stream()
                .map(campaign -> CampaignDataInfo.builder()
                        .communityId(campaign.getCommunityId())
                        .campaignGroupId(campaign.getCampaignGroupId())
                        .disabled(campaign.isDisabled())
                        .name(campaign.getName())
                        .build())
                .collect(Collectors.toList());

        // Build community DTO with campaign list
        CommunityShortInfo communityInfo = CommunityShortInfo.builder()
                .name(community.getName())
                .campaignDataList(campaignList)
                .build();

        communityList.add(communityInfo);
    }

    return communityList;
}

With Loop:

@Override
public List<CampaignShortData> getCampaigns(
        Workspace workspaceData,
        Community communityData,
        Member member,
        CampaignQueryParameters parameters
) {
    List<Long> campaignIds = findCampaignIds(
            communityData,
            mapCampaignQueryParameters(workspaceData, communityData, member, parameters)
    );

    List<CampaignShortData> campaigns = new ArrayList<>();

    if (campaignIds != null) {
        for (Long campaignId : campaignIds) {
            Optional<CampaignData> optionalCampaign = findCampaignById(communityData, campaignId);

            if (optionalCampaign.isPresent()) {
                CampaignData campaignData = optionalCampaign.get();

                CampaignShortData shortData = CampaignShortData.builder()
                        .communityId(campaignData.getCommunityId())
                        .campaignGroupId(campaignData.getCampaignGroupId())
                        .disabled(campaignData.isDisabled())
                        .name(campaignData.getName())
                        .build();

                campaigns.add(shortData);
            }
        }
    }

    return campaigns;
}

With Stream Api:

 @Override
    public List<CampaignShortData> getCampaigns(Workspace workspaceData, Community communityData, Member member, CampaignQueryParameters parameters) {
        List<Long> CampaignIds = findCampaignIds(communityData, mapCampaignQueryParameters(workspaceData, communityData, member, parameters));
        List<CampaignShortData> campaigns = new ArrayList<>();
        if (CampaignIds != null) {
            List<CampaignShortData> filteredCampaigns = CampaignIds.stream()
                    .map(campaignId -> findCampaignById(communityData, campaignId).orElse(null))
                    .filter(Objects::nonNull)
                    .map(campaignData ->  CampaignShortData.builder()
                            .communityId(campaignData.getCommunityId())
                            .campaignGroupId(campaignData.getCampaignGroupId())
                            .disabled(campaignData.isDisabled())
                            .name(campaignData.getName())
                            .build()
                    )
                    .toList();
            campaigns.addAll(filteredCampaigns);
        }
        return campaigns;
    }
